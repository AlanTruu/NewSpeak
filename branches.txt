func branch(num) {
    on (num < 1) {
        num = num + 1;
        redact num;
        return num;
    } otherwise {

        while (num < 100) {
            redact num;
            num = num + 1;
        }

        return num;
    }
}

var time_1 = time();
branch(0);
var time_2 = time();

var time_3 = time();
branch(1);
var time_4 = time();

var diff_1 = time_2 - time_1;
var diff_2 = time_4 - time_3;
redact diff_1;
redact diff_2;
